# Quick Hardware Test (hwtest_gui.py)
#
# No third-party Python packages are required.
# The script only uses the Python 3 standard library (including tkinter).
#
# --- Alpine packages (include these in the ISO image) ---
#   apk add python3 py3-tkinter             # Python + Tkinter (required)
#   apk add ffmpeg                          # webcam preview + MP3 speaker test
#   apk add alsa-utils                      # aplay, for speaker playback and the USB chime
#   apk add usbutils                        # lsusb, for USB checks and monitoring
#   apk add pulseaudio-utils                # optional paplay fallback if PulseAudio is configured
#
# --- Optional: only if you want to build a standalone executable ---
# The script detects frozen builds (sys.frozen), so it works with PyInstaller.
